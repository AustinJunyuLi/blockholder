\section{The general-equilibrium cutoff-shift channel: a region theorem and a counterexample}
\label{app:d8}

Recall the decomposition \eqref{eq:d1-decomp},
\[
\frac{\mathrm{d}\Delta^{\min}}{\mathrm{d}\kappa}
= \underbrace{E'(\kappa)}_{\text{(A) posterior/pricing}}
+ \underbrace{B(\kappa)}_{\text{(B) GE cutoff-shift}},
\qquad
B(\kappa)=\sum_{i\in\{1,0,D\}}\frac{\partial\Phi}{\partial k_i}\,\frac{\mathrm{d} k_i}{\mathrm{d}\kappa},
\]
with $\Delta^{\min}(\kappa)=\Phi(k(\kappa),\kappa)$ as in Lemma~\ref{lem:d1-ift}. Channel (A) is signed by Lemma~\ref{lem:d1-jensen} under \textup{(C*)}; the endpoints are pinned by Lemma~\ref{lem:endpoints}. This appendix controls channel (B). Throughout, $f$ denotes the signal density, $\omega=(\omega_E,\omega_H,\omega_Q,\omega_P)$ the action probabilities, and the seven feasible $(X,D)$ cells carry posteriors $\pi(X,D)$, prices $P(X,D)$, premia $\bar m(X,D)=m_0+(\mtil-m_0)\pi(X,D)$, and bid probabilities $p(X,D)$ from \eqref{eq:bid-prob}.

\subsection{The pricing layer is differentiable with explicit sensitivities}

\begin{lemma}[Pricing-layer IFT under A5]
\label{lem:d8-pricing}
Fix a cell $(X,D)$ and view the pricing fixed point \eqref{eq:pricing} as
$P=\delta\big[p(P,\bar m)\,(P+\bar m)+(1-p(P,\bar m))\,\hat V\big]$, an equation in $P$ given the cell inputs $u=(\pi,\hat V)$ with $\bar m=m_0+(\mtil-m_0)\pi$ and $\hat V=\E[v\mid X,D]+\Dtil\,\pi$. Under Assumption~\textup{(A5)} the feedback margin
\[
\ell(P,u)\;\equiv\;\delta\Big[p+\frac{\partial p}{\partial P}\,\big(P+\bar m-\hat V\big)\Big]
\]
satisfies $\ell<1$ along the equilibrium path, the solution $P^{*}(u)$ is unique and $C^{1}$, and
\[
\frac{\partial P^{*}}{\partial u_j}
=\frac{\delta\,\partial_{u_j}\big[p\,(P+\bar m)+(1-p)\hat V\big]\big|_{P}}{1-\ell(P^{*},u)},
\qquad
\Big|\frac{\partial P^{*}}{\partial u_j}\Big|
\;\le\;\frac{\delta\,\big|\partial_{u_j}[\cdot]\big|_{P}}{1-\bar\ell},
\]
for any uniform margin bound $\bar\ell<1$. In particular all cell objects $(P^{*},p,\bar m)$ are $C^{1}$ functions of $(\omega,\kappa)$ through the Bayes maps of Proposition~\ref{prop:posteriors}, with explicit derivatives by the chain rule.
\end{lemma}

\begin{proof}
$\Phi(\cdot)$ and $\varphi(\cdot)$ are smooth, so the right side of the fixed-point equation is $C^{1}$ in $(P,u)$. A uniform margin $\bar\ell<1$ along the path is part of the maintained Assumption~\textup{(A5a)}: the body's sufficient condition $\delta/\sigma_\xi<1/\varphi(0)$ bounds only the feedback component of $\ell$, not the direct term $\delta p$ (Remark~\ref{rem:A5margins} in the body makes exactly this point), but the \emph{signed} $\ell$ is small at the sampled equilibrium cells ($\ell\in[0.03,0.43]$; the value gap $P+\bar m-\hat V$ enters with the sign that shrinks $\ell$ there); the implicit function theorem then yields uniqueness, $C^{1}$-ness, and the displayed quotient, whose denominator is $1-\ell$. The Bayes maps $\omega\mapsto\pi(X,D)$ are rational functions with positive denominators on reached cells (full-support noise), hence $C^{1}$; composition gives the final statement.
\end{proof}

\subsection{The cutoff layer: operator bound from the contraction modulus}

Write the equilibrium system as $F(k,\kappa)=k-T(k,\kappa)=0$, where $T$ is the best-response map of Section~\ref{sec:equilibrium} (the indifference-point update whose damped iteration the solver implements, with prices and posteriors evaluated equilibrium-consistently at $(k,\kappa)$ via Lemma~\ref{lem:d8-pricing}).

\begin{lemma}[Cutoff-layer IFT with an inversion-free bound]
\label{lem:d8-cutoff}
Suppose Assumption~\textup{(A6)} holds in the quantified form: $T(\cdot,\kappa)$ is $C^{1}$ and $\|D_kT(k,\kappa)\|_{\infty}\le L<1$ in a neighbourhood of the equilibrium path. Then $k(\kappa)$ is $C^{1}$,
\[
\frac{\mathrm{d} k}{\mathrm{d}\kappa}
=\big(I-D_kT\big)^{-1}\,\partial_\kappa T ,
\qquad
\Big\|\frac{\mathrm{d} k}{\mathrm{d}\kappa}\Big\|_{\infty}
\;\le\;\frac{\|\partial_\kappa T(k(\kappa),\kappa)\|_{\infty}}{1-L},
\]
and hence, with weights $W_i(\kappa)\equiv|\partial\Phi/\partial k_i|$,
\begin{equation}
|B(\kappa)|\;\le\;\bar B(\kappa)\;\equiv\;\Big(\sum_{i}W_i(\kappa)\Big)\,
\frac{\|\partial_\kappa T(k(\kappa),\kappa)\|_{\infty}}{1-L}.
\label{eq:d8-Bbar}
\end{equation}
Every factor in $\bar B$ is computable from a single equilibrium solve at $\kappa$ plus one finite-difference column $\partial_\kappa T$; no Jacobian inversion and no knowledge of the true path derivative $\mathrm{d} k/\mathrm{d}\kappa$ is required.
\end{lemma}

\begin{proof}
$D_kF=I-D_kT$ with $\|D_kT\|_\infty\le L<1$, so $I-D_kT$ is invertible with the Neumann bound $\|(I-D_kT)^{-1}\|_{\infty}\le\sum_{n\ge0}L^{n}=1/(1-L)$. The IFT gives $C^{1}$-ness of $k(\kappa)$ and $\mathrm{d} k/\mathrm{d}\kappa=-(D_kF)^{-1}\partial_\kappa F=(I-D_kT)^{-1}\partial_\kappa T$ (using $\partial_\kappa F=-\partial_\kappa T$). Submultiplicativity gives the displayed bound, and \eqref{eq:d8-Bbar} follows from the triangle inequality on $B$. The computability claim is by inspection: $W_i$ from Lemma~\ref{lem:d8-weights} below at the solved $(k(\kappa),\kappa)$, the modulus $L$ from a finite-difference $D_kT$ (or any certified upper bound), and $\partial_\kappa T$ from one $\kappa$-perturbed best-response evaluation at fixed $k$.
\end{proof}

Relative to Lemma~\ref{lem:d1-ift}, whose bound $|B|\le(\sum_i|\partial\Phi/\partial k_i|)\|\mathrm{d} k/\mathrm{d}\kappa\|_\infty$ requires the true equilibrium derivative along the whole path, \eqref{eq:d8-Bbar} replaces that object with the contraction modulus already maintained by Assumption~\textup{(A6)}. This is what converts the transfer inequality from an ex-post diagnostic into an ex-ante checkable hypothesis.

\subsection{The weights: explicit structure and conditional signs}

\begin{lemma}[Weight decomposition]
\label{lem:d8-weights}
Write $\Phi(k,\kappa)=\sum_{(X,D)}\mu_{X,D}(k,\kappa)\,\bar m(X,D)\,p(X,D)$, where $\mu_{X,D}$ is the cell mass induced by $(\omega(k),\kappa)$ and $(\bar m,p)$ are evaluated at the cell objects of Lemma~\ref{lem:d8-pricing}. Then for $i\in\{1,0,D\}$,
\begin{equation}
\frac{\partial\Phi}{\partial k_i}
=\underbrace{f(k_i)\sum_{(X,D)}\frac{\partial\mu_{X,D}}{\partial\omega}\cdot e_i\;\bar m\,p}_{\textup{boundary (mass-shift) term}}
\;+\;\underbrace{\sum_{(X,D)}\mu_{X,D}\,(\mtil-m_0)\Big[p+\bar m\,\frac{\partial p}{\partial\bar m}\Big]\frac{\partial\pi(X,D)}{\partial k_i}}_{\textup{inference term}}
\;+\;\underbrace{\sum_{(X,D)}\mu_{X,D}\,\bar m\,\frac{\partial p}{\partial P}\,\frac{\partial P^{*}}{\partial k_i}}_{\textup{pricing term}},
\label{eq:d8-weights}
\end{equation}
where $e_i$ is the signed pair of adjacent-region mass shifts at the boundary $k_i$ (raising $k_1$ moves $f(k_1)\,\mathrm{d} k$ of mass from Hold to Exit; $k_0$ from Quiet to Hold; $k_D$ from Public to Quiet), $\partial\pi/\partial k_i$ and $\partial P^{*}/\partial k_i$ are the Bayes and pricing chain-rule derivatives through $\omega(k)$, and $\partial p/\partial\bar m=-\varphi(t)/\sigma_\xi$, $\partial p/\partial P=-\varphi(t)/\sigma_\xi$ with $t=(\bar m+K-\bar S+P)/\sigma_\xi$.
All terms are explicit compositions of $(f,\Phi$-cdf differences$,\varphi)$ and the equilibrium cell objects. Numerically, the script evaluates the weights $W_i$ by central finite differences of $\Phi$ in each cutoff (the decomposition above guarantees these derivatives exist and are continuous) and spot-validates the pricing-layer quotient of Lemma~\ref{lem:d8-pricing} cell by cell.
\end{lemma}

\begin{proof}
Differentiate the finite sum term by term. Cell masses depend on $k$ only through $\omega(k)$, whose components are Gaussian probabilities of the action regions, giving the boundary terms with density factor $f(k_i)$ and adjacent-region transfer $e_i$ (the noise mixture is $k$-free). The remaining dependence is through $\bar m(\pi)$ and $p(\bar m,P)$: the product rule on $\bar m\,p$ gives the inference bracket, and Lemma~\ref{lem:d8-pricing} gives the pricing term with $\partial P^{*}/\partial k_i$ well defined. Finiteness and continuity are inherited from the smooth ingredients.
\end{proof}

\begin{remark}[Conditional signs, in the spirit of monotone comparative statics]
\label{rem:d8-signs}
Two cell-level sign statements are available conditionally; we use neither as an input to Theorem~\ref{thm:d8-region}, and we record their status at the baseline equilibrium honestly (direct computation at the $\kappa=0.5$ solve; these are \emph{not} certified by the check script):
(i) \emph{Deterrence moderation.} The inference bracket in \eqref{eq:d8-weights} is positive iff $p>\bar m\,\varphi(t)/\sigma_\xi$ -- the cell's bid probability exceeds its deterrence slope. Note Remark~\ref{rem:d5-anegative}'s $a(\pi)<0$ is derived under the surrogate $P(\pi)=\bar m(\pi)$, not equilibrium prices; at the actual baseline equilibrium $t>0$ on most reached cells, and the bracket condition \emph{fails} on the disclosed ($\pi=1$) cells --- harmlessly, since $\partial\pi/\partial k_i=0$ there at the collapsed baseline, so those cells carry no inference weight. On the two operative inferred cells the condition holds with margins $+0.38$ at $(X,D)=(-1,0)$ and $+0.12$ at $(0,0)$. It is the single-crossing condition under which the cell contribution $\bar m\,p$ increases in the posterior $\pi$ (Milgrom--Shannon, applied to the cell payoff in $(\pi;\sigma_\xi)$).
(ii) \emph{Boundary-term signs.} $e_D$ moves mass from the disclosed branch ($\pi=1$, $\kappa$-invariant by Appendix~\ref{app:proof-disclosed-invariance}) to the inferred $\bar\pi$-atom, so its sign is that of $\big[\bar m\,p\big]_{\bar\pi\text{-cells}}-\big[\mtil\,\bar p_1\big]$. At the baseline this difference is \emph{positive} (per unit mass, $\approx0.074$ vs $\approx0.0037$): disclosure deters bids through the price run-up, so moving mass from Public to Quiet \emph{raises} the premium-weighted bid mass --- the opposite of the na\"ive ``disclosure carries the premium with certainty'' intuition. $e_1$ shifts Exit/Hold composition and operates purely through $\bar\pi$.
These signs can sharpen the bound $\bar B$ cell by cell where they hold but are not needed for Theorem~\ref{thm:d8-region}.
\end{remark}

\subsection{The region theorem: GE-dominance as a checkable hypothesis}

\begin{theorem}[Strict single peak, up to $\varepsilon$-localization, on a certified region]
\label{thm:d8-region}
Maintain Assumptions \textup{A1--A7}, the quantified contraction of Lemma~\ref{lem:d8-cutoff}, and \textup{(C*)} on $[\underline\kappa,\overline\kappa]$. Let $E(\kappa)$ denote channel (A): the antiderivative, along the equilibrium path $k(\kappa)$, of the frozen-cutoff derivative $E'(\kappa)=\partial_\kappa\Phi(k,\kappa)\big|_{k=k(\kappa)}$. Suppose additionally
\begin{enumerate}
\item[\textup{(R0)}] \emph{channel (A) single-peaked:} $E'$ is positive before and negative after a unique interior crossing $\kappa_A^{\dagger}\in(\underline\kappa,\overline\kappa)$. (Under \textup{(C*)}, Lemma~\ref{lem:d1-jensen} delivers an interior maximum of $E$ strictly above the endpoint values; the single-crossing form is the additional regularity recorded as \textup{(NR1)} in the D1 record, and is verified here directly by the script's sign-change count of the channel-(A) profile.)
\end{enumerate}
and that there is $\varepsilon>0$ such that
\begin{enumerate}
\item[\textup{(R1)}] \emph{pointwise dominance off the peak:}\; $\bar B(\kappa)<|E'(\kappa)|$ for all $\kappa\in[\underline\kappa,\overline\kappa]\setminus(\kappa_A^{\dagger}-\varepsilon,\kappa_A^{\dagger}+\varepsilon)$;
\item[\textup{(R2)}] \emph{integral control through the peak:}\; $\int_{\kappa_A^{\dagger}-\varepsilon}^{\kappa_A^{\dagger}+\varepsilon}\bar B(\kappa)\,\mathrm{d}\kappa\;<\;\min\Big\{E(\kappa_A^{\dagger}-\varepsilon),\,E(\kappa_A^{\dagger}+\varepsilon)\Big\}-\max\Big\{E(\underline\kappa),E(\overline\kappa)\Big\}$.
\end{enumerate}
Then $\Delta^{\min}$ is strictly increasing on $[\underline\kappa,\kappa_A^{\dagger}-\varepsilon]$, strictly decreasing on $[\kappa_A^{\dagger}+\varepsilon,\overline\kappa]$, and attains its global maximum at an interior $\kappa^{\dagger}\in(\kappa_A^{\dagger}-\varepsilon,\kappa_A^{\dagger}+\varepsilon)$. In particular $\mathrm{d}\Delta^{\min}/\mathrm{d}\kappa$ changes sign exactly once \emph{outside} the $\varepsilon$-ball, with any further sign changes (an even number on each side of the maximizer) confined to the ball. The conclusion of Hypothesis~\ref{hyp:d1-cutoffshift} is thereby discharged on $[\underline\kappa,\overline\kappa]$ in the \emph{$\varepsilon$-localized} form: $\Delta^{\min}$ is single-peaked up to $\varepsilon$-localization of its peak; the pointwise single-crossing of the derivative \emph{inside} the ball is not claimed (and does not follow from \textup{(R1)}--\textup{(R2)}, which control $B$ there only in integral).
\end{theorem}

\begin{proof}
Off the $\varepsilon$-ball, $|B|\le\bar B<|E'|$ by (R1) and Lemma~\ref{lem:d8-cutoff}, so $\mathrm{d}\Delta^{\min}/\mathrm{d}\kappa=E'+B$ carries the sign of $E'$: positive on $[\underline\kappa,\kappa_A^{\dagger}-\varepsilon]$ (where $E'>0$ by (R0)), negative on $[\kappa_A^{\dagger}+\varepsilon,\overline\kappa]$. Hence $\Delta^{\min}$ is strictly monotone on the two outer segments, any maximizer lies in the closed $\varepsilon$-ball, and the derivative has exactly one sign change outside the ball. The ball's edges are not maximizers: (R1) holds \emph{at} $\kappa_A^{\dagger}\pm\varepsilon$ (they lie in the off-ball set as defined), so $\Delta^{\min}{}'(\kappa_A^{\dagger}-\varepsilon)>0$ and $\Delta^{\min}{}'(\kappa_A^{\dagger}+\varepsilon)<0$, and $\Delta^{\min}$ is $C^{1}$ by Lemma~\ref{lem:d8-cutoff}; hence the global maximizer is \emph{interior} to the ball. (R2) supplies level control: integrating $\Delta^{\min}{}'=E'+B$ from either ball edge, the ball's accumulated $\bar B$-mass is too small to push any ball value below the outer-segment endpoints, so the maximum over the ball strictly exceeds $\Delta^{\min}(\underline\kappa)$ and $\Delta^{\min}(\overline\kappa)$ --- the interior maximizer is global. Inside the ball (R1) is silent and (R2) bounds $B$ only in integral, so pointwise sign changes of the derivative there cannot be excluded; the conclusions are exactly as stated. (As $\varepsilon\downarrow0$ under continuity of $\bar B$ the localization tightens to $\kappa_A^{\dagger}$.)
\end{proof}

\begin{corollary}[Baseline instantiation -- certified interval]
\label{cor:d8-baseline}
At the baseline calibration the script \texttt{d8\_ge\_dominance\_check.py} evaluates $L$, $\|\partial_\kappa T\|_\infty$, the weights of Lemma~\ref{lem:d8-weights}, and the channel-(A) profile on $\kappa\in[0.30,0.85]$ (23 grid points, spacing $0.025$; the statements below are grid-point certificates with a continuity appeal between nodes, not interval arithmetic), and reports (JSON artifact): contraction modulus $L\le0.836<1$ at every node; channel-(A) peak at $\kappa\approx0.60$ with a single sign change of $E'$ on the grid (\textup{(R0)}); \textup{(R1)} holding at \emph{every} off-ball grid point and \textup{(R2)} with two orders of magnitude to spare (ball integral $2.5\times10^{-5}$ against margin $4.7\times10^{-3}$) under the exact-IFT bound, certifying $[0.30,0.85]$ --- a span that is \emph{grid-limited, not failure-limited} (the exact bound also passes beyond both ends of the tested grid); and -- the operative result -- the \emph{inversion-free} bound \eqref{eq:d8-Bbar} alone certifying $[0.35,0.825]$, whose endpoints \emph{are} genuinely bracketed by failures (R1 fails at $0.325$ and $0.85$ under the loose bound). On these intervals the equilibrium hump is a theorem in the $\varepsilon$-localized sense of Theorem~\ref{thm:d8-region}, conditional on the maintained Assumptions \textup{A1--A7}, \textup{(C*)}, and the grid-verified \textup{(R0)}; Hypothesis~\ref{hyp:d1-cutoffshift} is discharged there in that sense and remains a (falsifiable) regularity in the residual endpoint neighbourhoods. Two further notes for honesty: the entire baseline path sits on the collapsed-Hold manifold ($k_{0}=k_{1}$ at all 23 nodes), so the IFT machinery is applied to the effective two-cutoff system, with persistence of the collapse observed at every node; and the loose-bound margins at the certified boundary are thin ($2.9\%$ at $0.35$, $6.1\%$ at $0.825$), so sub-grid violations there, while not found in finer probes, are not formally excluded.
\end{corollary}

\subsection{The counterexample: GE-dominance is false as a global claim}
\label{app:d8-counter}

\begin{proposition}[A certified trough with channel (A) single-peaked]
\label{prop:d8-counter}
Consider the robustness cells of the calibration family with $\sigma_\xi=0.60$ (Appendix~\ref{app:d5}, Remark~\ref{rem:d5-conditional}). There: (i) the curvature condition holds everywhere -- indeed it cannot fail on the family (Remark~\ref{rem:d5-vacuous} gives $a(\pi)<0$, hence $p'<0$ and $p''<0$, so both the cell payoff $g(\pi)=\bar m(\pi)p(\pi)$ and Lemma~\ref{lem:d1-jensen}'s object $h(\pi)=\pi\,p(\pi)$, with $h''=2p'+\pi p''<0$, are concave: \textup{(C*)} holds) -- so channel (A) has an interior maximum above equal endpoints by Lemma~\ref{lem:d1-jensen}, and the script verifies the single-crossing form \textup{(R0)} cell by cell; (ii) the full-equilibrium profile $\Delta^{\min}(\kappa)$ exhibits an interior \emph{trough}: the script certifies troughs at $(\sigma_\xi,\bar S)=(0.60,1.44),(0.60,1.54),(0.60,1.64)$ with interior minimizers $\kappa\in\{0.275,0.35,0.475\}$ and depths up to $2.8\times10^{-3}$, channel (A) single-peaked in every cell, and $\int|B|$ exceeding the channel-(A) amplitude in every cell (e.g.\ $0.021$ vs $0.0069$ at $\bar S=1.64$); hence (iii) $B(\kappa)$ overturns channel (A) on these cells, and the conclusion of Hypothesis~\ref{hyp:d1-cutoffshift} \emph{fails}. Consequently no proof of the unconditional hump can dispense with a restriction on the GE channel: the region hypothesis of Theorem~\ref{thm:d8-region} is not removable, and at the $\sigma_\xi=0.60$ cells the theorem's certification \emph{must} fail --- given (i), (ii), and the trough, no $(\varepsilon,\textup{R1},\textup{R2})$ configuration can hold there whenever the contraction and \textup{(R0)} hypotheses do (modus tollens on Theorem~\ref{thm:d8-region}).
\end{proposition}

\begin{proof}
(i) is Remark~\ref{rem:d5-vacuous} plus the concavity bridge displayed in the statement and Lemma~\ref{lem:d1-jensen}. (ii) is the certified computation: the script re-solves the cells at the standard residual gate ($5\times10^{-3}$) on the solver's reference path, evaluates $\Delta^{\min}$ on a fine $\kappa$ grid, computes channel (A) by $\kappa$-perturbation at frozen cutoffs and $B$ as the residual of the total derivative, and reports an interior minimum with $\int|B|$ exceeding the channel-(A) amplitude (JSON artifact). (iii) follows from (i)+(ii) by the decomposition \eqref{eq:d1-decomp}: with equal endpoints and (A) single-peaked in the single-crossing sense of \textup{(R0)}, a trough requires $E'+B$ to change sign in the opposite pattern, impossible unless $|B|>|E'|$ on a set of positive measure. The final claims are immediate.
\end{proof}

\begin{remark}[Economic boundary: where and why GE feedback wins]
\label{rem:d8-boundary}
Channel (A)'s amplitude is a deterrence object: it is the chord gap of the concave cell payoff $g(\pi)=\bar m(\pi)p(\pi)$ (Lemma~\ref{lem:d1-jensen}'s $h(\pi)=\pi\,p(\pi)$ is concave under the same condition), whose scale is governed by the bid-incidence sensitivity $|\partial p/\partial\bar m|=\varphi(t)/\sigma_\xi$. As $\sigma_\xi$ grows, entry becomes insensitive to the activism premium, the chord gap flattens, and the fixed-cutoff hump shrinks -- while the cutoff system's response to $\kappa$ (the $B$ weights, driven by trading rents and disclosure composition) does not shrink in step. GE feedback therefore dominates precisely in the high-$\sigma_\xi$ corner, which is where the troughs live. The testable content: the liquidity hump in minority gains should be strongest where takeover entry is most sensitive to expected resistance (low effective $\sigma_\xi$), and can invert where entry is premia-insensitive.
\end{remark}

\subsection{What this buys downstream}
\label{app:d8-downstream}

The machinery applies verbatim to the welfare objects: $W_B$ and $W_{\textup{bid}}$ have their own weight sums $W^{(B)}_i$, $W^{(\textup{bid})}_i$ (computed by the same decomposition as Lemma~\ref{lem:d8-weights}), so the bundle $\mathcal B(\kappa)$ of Proposition~\ref{prop:planner-wedge} obeys $|\mathcal B(\kappa)-\partial_\kappa W_B|_{k}-\mathrm{d} W_{\textup{bid}}/\mathrm{d}\kappa|\le(\sum_i W^{(B)}_i)\,\|\partial_\kappa T\|_\infty/(1-L)$, putting a computable band around the liquidity-planner wedge of Proposition~\ref{prop:planner-wedge} and the disclosure-planner terms of Proposition~\ref{prop:disclosure-wedge}. We do not re-derive the welfare propositions here; the point is that their conditional clauses (\textup{(W*)} and the unsigned GE terms) inherit the same certify-or-falsify structure, replacing "unsigned" with "signed up to a computable band."

\subsection{Ledger}
\label{app:d8-ledger}

\emph{Proved:} Lemmas \ref{lem:d8-pricing}--\ref{lem:d8-weights}, Theorem~\ref{thm:d8-region}, and Proposition~\ref{prop:d8-counter}(iii) given (ii). \emph{Computable-verified at the calibration (JSON artifacts):} the contraction modulus $L$; the certified interval of Corollary~\ref{cor:d8-baseline} with margins; the $\sigma_\xi=0.60$ trough and its channel decomposition. \emph{Open:} a calibration-free analytic modulus for \textup{(A6)}; a global analytic sign for $B$ (now known to be impossible in the strong form: Proposition~\ref{prop:d8-counter}); endpoint neighbourhoods (both $E'$ and $B$ degenerate at the exact grid edges, as flagged in Appendix~\ref{app:proof-d1-GE}).
