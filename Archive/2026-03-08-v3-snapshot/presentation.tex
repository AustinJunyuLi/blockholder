\begin{frame}{Motivation and Gap}
\hypertarget{main:motivation}{}
\hypertarget{main:lit}{}
\small
\begin{columns}[T,onlytextwidth]
\column{0.55\textwidth}
\textbf{Three empirical facts motivate the paper.}
\begin{itemize}\itemsep0.45em
  \item \citealp{BravJiangPartnoyThomas2008}: activist campaigns often culminate in corporate control events.
  \item \citealp{GreenwoodSchor2009}: a large share of activist returns comes through takeover outcomes.
  \item \citealp{BackEtAl2018}: pre-filing trading suggests activists exploit informational advantages before disclosure.
\end{itemize}

\vspace{0.3em}
\begin{alertblock}{Gap}
No existing framework combines \textbf{endogenous exit vs.\ voice choice}, \textbf{order-flow inference}, and \textbf{takeover entry from public market signals}.
\end{alertblock}

\column{0.41\textwidth}
\begin{block}{Question}
How does market liquidity shape \emph{takeover premia} for minority shareholders when a blockholder can \emph{exit}, \emph{hold}, or \emph{engage}?
\end{block}

\vspace{0.25em}
\begin{block}{Closest inspiration}
\citealp{EdmansGoldsteinJiang2015}: public market information feeds back into takeover outcomes. This paper adds endogenous exit/voice choice and disclosure.
\end{block}

\vspace{0.2em}
\textbf{Bridge:} exit / voice, microstructure, and corporate control.
\end{columns}
\end{frame}

% -- SLIDE 3: Main mechanism -------------------------------------------------
\begin{frame}{Main Mechanism}
\hypertarget{main:mechanismpreview}{}
\small
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{What the market learns}
\begin{itemize}\itemsep0.35em
  \item Before activism is fully public, trading and disclosure leave a public footprint.
  \item Without a filing, order flow makes activism look more or less likely.
  \item Liquidity matters because it changes how informative that footprint is.
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{Why takeover outcomes move}
\begin{itemize}\itemsep0.35em
  \item When activism looks more likely, expected standalone value is higher.
  \item Expected takeover premia are also higher.
  \item But some bidders stay out because the target looks harder or more expensive to acquire.
  \item Minority takeover gains reflect the balance of those forces.
\end{itemize}
\end{block}
\end{columns}

\vspace{0.45em}
\textbf{In short:} the same public footprint that raises expected value and premia can also discourage bidder entry, and takeover gains reflect that balance.

\begin{itemize}\itemsep0.35em
  \item Blockholder strategy is a monotone cutoff rule \choice{$\sigma_B(s)$}, summarized by \choice{$k_1\le k_0\le k_D$}.

\begin{itemize}\itemsep0.35em
  \item The market maker sets prices from those inferences.
  \item The bidder decides whether to enter using the same public signal and private synergy \obs{$\xi$}.
  \item The cutoff rule must be optimal given those pricing and bidding responses.
  \item Beliefs, prices, bids, and cutoffs must fit together in a self-consistent way.
\end{itemize}
\end{block}

\begin{frame}{Inference, Bid Entry, and Prices}
\hypertarget{main:engagement}{}
\hypertarget{main:prop2}{}
\hypertarget{main:pricedecomp}{}
\hypertarget{main:prices}{}
\small
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\textbf{Inference}
\begin{itemize}\itemsep0.35em
  \item Disclosed branch: \obs{$D=1$} $\Rightarrow$ \outcome{$\pi(X,1)=1$}.
  \item Nondisclosed branch: \obs{$D=0$} mixes Exit, Hold, and Quiet Voice through order flow \obs{$X$}.
  \item Liquidity \obs{$\kappa$} matters by changing how informative \obs{$X$} is in nondisclosed states.
  \item Channel: \obs{$X$} changes \outcome{$\pi$}, which moves \outcome{$\bar m$}, \outcome{$p$}, and \outcome{$\Delta^{\min}$}.
\end{itemize}

\column{0.48\textwidth}
\textbf{Bidder and prices}
{\footnotesize
\[
\outcome{\hat V}=\E[v\mid \obs{X},\obs{D}]+\tilde{\Delta}\,\outcome{\pi},
\qquad
\outcome{\bar m}=m_0+(\tilde m-m_0)\outcome{\pi}
\]
\[
\outcome{p}=\lambda_B\!\big(1-\Lambda(\Gamma)\big),
\qquad
\Gamma=\frac{\outcome{\hat V}+\outcome{\bar m}+K-(\bar S+\outcome{\pi}\Delta_S)}{s_{\xi}}
\]
\[
\outcome{P_{\textup{post}}}=\delta\big(\outcome{\hat V}+\outcome{p}\,\outcome{\bar m}\big),
\qquad
\outcome{P_{\textup{trade}}}(\obs{X})=\E[\outcome{P_{\textup{post}}}\mid \obs{X}]
\]
}
\begin{itemize}\itemsep0.3em
  \item Under A5 (net deterrence), higher inferred engagement raises value and premia enough to deter marginal bids.
  \item Sequential mechanism: public signal $\to$ beliefs $\to$ bidder entry $\to$ prices.
\end{itemize}
\end{columns}

\begin{frame}{B1. How do you establish the cutoff ordering?}
\hypertarget{backup:B1}{}
\footnotesize
\begin{alertblock}{Proof Sketch for Proposition 1}
Key ingredients:
\begin{itemize}\itemsep0.15em
  \item Payoffs are affine in $\hat v(\choice{s})=\E[v\mid \choice{s}]$; $\hat v$ increases in \choice{$s$}
  \item Engagement cost \choice{$C(s)$} is (weakly) decreasing in \choice{$s$}
  \item \outcome{$P_{\textup{trade}}(X)$} is independent of \choice{$s$} within a given action
\end{itemize}
These imply strict single-crossing of adjacent payoff differences, yielding a monotone best response with cutoffs \choice{$k_1\le k_0\le k_D$}.
\end{alertblock}

\begin{alertblock}{Baseline action-set interpretation}
The model is written for a control-oriented activist, so the only threshold-crossing action in the baseline action set is Public Voice. Appendix lemma: passive threshold crossing is dominated for sufficiently high signals, and in the baseline calibration \choice{$k_D\ge\bar{s}$}.
\end{alertblock}

\end{frame}

% -- B2 ----------------------------------------------------------------------
\begin{frame}{B2. Existence and numerical fixed-point selection}
\hypertarget{backup:B2}{}
\small
\begin{alertblock}{Existence (Proposition 4)}
Existence is stated under maintained regularity conditions: a compact ordered cutoff set $\Theta$, a well-defined cutoff mapping $T$, and continuity on reached information sets with an off-path completion. Brouwer then implies a fixed point.
\end{alertblock}

\begin{alertblock}{Uniqueness (Assumption A6)}
No analytic uniqueness theorem is claimed. The solver converges to the same fixed point from multiple starts in all reported exercises.
\end{alertblock}

\begin{alertblock}{Solver Implementation}
Damped fixed-point iteration with \texttt{scipy.optimize.brentq} for sub-problems. Multi-start search with collapsed-hold fallback. Convergence tolerance: $10^{-6}$.
\end{alertblock}

\end{frame}

% -- B2a ---------------------------------------------------------------------
\begin{frame}{B2a. Full monotone-cutoff PBE definition}
\hypertarget{backup:B2a}{}
\footnotesize
\begin{alertblock}{Equilibrium object}
A monotone-cutoff Perfect Bayesian Equilibrium is a tuple
\[
\Big(\choice{k_1,k_0,k_D},\; \outcome{\mu(\cdot\mid h)},\; \outcome{P_{\textup{trade}}(X)},\; \outcome{P_{\textup{post}}(h)},\; \outcome{\tilde p(h,\xi)}\Big)
\]
such that the following conditions hold.
\end{alertblock}

\begin{enumerate}\itemsep0.35em
  \item \textbf{Blockholder optimality:} the cutoff rule induced by \choice{$(k_1,k_0,k_D)$} maximizes expected payoff at every signal \choice{$s$}.
  \item \textbf{Bidder best response:} for every reached public state \obs{$h$} and private synergy \obs{$\xi$}, entry follows the surplus-maximizing rule \outcome{$\tilde p(h,\xi)$}.
  \item \textbf{Competitive pricing:} \outcome{$P_{\textup{trade}}(X)=\delta\E[Y\mid X]$} and \outcome{$P_{\textup{post}}(h)=\delta\E[Y\mid h]$}.
  \item \textbf{Belief consistency:} \outcome{$\mu(\cdot\mid h)$} is Bayes-consistent on each reached public state.
  \item \textbf{Cutoff indifference:} adjacent-action indifference equalities hold at \choice{$k_1,k_0,k_D$}.
\end{enumerate}

\returnlink{main:eqmdef}{Slide 6: How the Model Closes}
\end{frame}

% -- B2b ---------------------------------------------------------------------
\begin{frame}{B2b. Reached public states and Bayes consistency}
\hypertarget{backup:B2b}{}
\footnotesize
\textbf{Public state.}
Let \obs{$h$} denote the public information state at $t=1.5$. In the baseline, \obs{$h$} is indexed by \obs{$(X,D)$}.

\vspace{0.35em}
\begin{alertblock}{On-path belief formula}
For any reached public state \obs{$h$} with \obs{$\PP(h)>0$},
\[
\outcome{\mu(a\mid h)}=\frac{\PP(a,h)}{\PP(h)}.
\]
Equivalently, beliefs over Exit / Hold / Quiet Voice / Public Voice are pinned down by the cutoff strategy and the noise distribution via Bayes' rule.
\end{alertblock}

\textbf{Reached states in the baseline}
\begin{itemize}\itemsep0.3em
  \item Disclosed branch: \obs{$D=1$} implies Public Voice, so only \obs{$(0,1),(1,1),(2,1)$} can occur.
  \item Nondisclosed branch: \obs{$D=0$} mixes Exit, Hold, and Quiet Voice through order flow.
  \item Off-path states require a completion for existence, but the quantitative objects use on-path beliefs.
\end{itemize}

\returnlink{main:eqmdef}{Slide 6: How the Model Closes}
\end{frame}

\begin{frame}{B2c. What does the bidder observe?}
\hypertarget{backup:B2c}{}
\footnotesize
\begin{alertblock}{Reduced-form interpretation}
The bidder need not literally observe raw exchange-level order flow. What the theorem package requires is that, by the entry stage, both the market maker and the bidder condition on the same \emph{public state} \obs{$h$}.
\end{alertblock}

\begin{itemize}\itemsep0.35em
  \item In the baseline, \obs{$h$} is indexed by the public trading/disclosure outcome \obs{$(X,D)$}.
  \item This is a sufficient-statistic formulation: any richer public history generating the same posterior \outcome{$\mu(\cdot\mid h)$} is observationally equivalent for prices and bidder entry.
  \item The economics is public learning from a trading footprint, not literal tape-reading by bidders.
\end{itemize}

\begin{frame}{B3. Conditional means and truncated normal expectations}
\hypertarget{backup:B3}{}
\small
Let $\alpha_i=(\choice{k_i}-\mu)/\sigma_s$ and inverse Mills ratios
$\lambda_L(\alpha)=\phi(\alpha)/\Phi(\alpha)$,
$\lambda_U(\alpha)=\phi(\alpha)/(1-\Phi(\alpha))$.

\begin{alertblock}{Conditional Means (Lemma 3)}
\begin{align*}
\mu_E&=\E[v\mid \choice{s}<\choice{k_1}]=\mu-\beta\sigma_s\lambda_L(\alpha_1), &
\mu_P&=\E[v\mid \choice{s}\ge \choice{k_D}]=\mu+\beta\sigma_s\lambda_U(\alpha_D),\\[0.3em]
\mu_H&=\mu+\beta\sigma_s\frac{\phi(\alpha_1)-\phi(\alpha_0)}{\Phi(\alpha_0)-\Phi(\alpha_1)}, &
\mu_Q&=\mu+\beta\sigma_s\frac{\phi(\alpha_0)-\phi(\alpha_D)}{\Phi(\alpha_D)-\Phi(\alpha_0)}.
\end{align*}
\end{alertblock}

These enter \outcome{$\hat V(X,D)$} through the conditional expectation $\E[v\mid \obs{X},\obs{D}]$, which weights $\mu_E$, $\mu_H$, $\mu_Q$, $\mu_P$ by the posterior action probabilities.

\end{frame}

% -- B5 ----------------------------------------------------------------------
\begin{frame}{B5. Mechanism and pricing logic}
\hypertarget{backup:B5}{}
\small
\begin{alertblock}{Step 1: Public signal to activism beliefs}
Trading and disclosure generate the public signal \obs{$(\obs{X},\obs{D})$}. The market maker and bidder update to the posterior activism belief \outcome{$\pi(\obs{X},\obs{D})=\PP(\choice{a}=1\mid \obs{X},\obs{D})$}.
\begin{itemize}\itemsep0.2em
  \item \obs{$D=1$} reveals Public Voice, so \outcome{$\pi(\obs{X},1)=1$}.
  \item \obs{$D=0$} mixes Exit, Hold, and Quiet Voice through order flow \obs{$X$}.
\end{itemize}
\end{alertblock}

\begin{alertblock}{Step 2: Beliefs move value, premia, and bid incentives}
\[
\outcome{\hat V}(\obs{X},\obs{D})=\E[v\mid \obs{X},\obs{D}]+\tilde\Delta\,\outcome{\pi}(\obs{X},\obs{D}),
\qquad
\outcome{\bar m}(\obs{X},\obs{D})=m_0+(\tilde m-m_0)\outcome{\pi}(\obs{X},\obs{D}).
\]
\[
\outcome{p}(\obs{X},\obs{D})=\lambda_B(1-\Lambda(\Gamma)),\qquad
\Gamma=\frac{\outcome{\hat V}+\outcome{\bar m}+K-(\bar S+\outcome{\pi}\Delta_S)}{s_\xi}.
\]
Higher inferred activism raises expected value and premia; under A5 it also raises the entry cutoff \obs{$\Gamma$} and deters marginal bids.
\end{alertblock}

\vfill
\returnlink{main:engagement}{Slide 9: Inference, Bids, Prices}\hfill
\backuplink{backup:B5b}{Pricing step}
\end{frame}

\begin{frame}{B5b. Pricing from the mechanism}
\hypertarget{backup:B5b}{}
\footnotesize
\begin{alertblock}{Step 3: Post-disclosure price}
Conditional on the realized public signal, competitive pricing averages standalone value and expected takeover surplus:
\[
\outcome{P_{\textup{post}}}(\obs{X},\obs{D})=
\delta\big(\outcome{\hat V}(\obs{X},\obs{D})+\outcome{p}(\obs{X},\obs{D})\outcome{\bar m}(\obs{X},\obs{D})\big).
\]
\end{alertblock}

\begin{alertblock}{Trade-date price}
Before disclosure is realized, the trade-date price averages over the later disclosure branch:
\[
\outcome{P_{\textup{trade}}}(\obs{X})=
\sum_{\obs{d}\in\{0,1\}}\PP(\obs{D}=\obs{d}\mid \obs{X})\,\outcome{P_{\textup{post}}}(\obs{X},\obs{d}).
\]
\end{alertblock}

\begin{alertblock}{Timing logic}
\begin{itemize}\itemsep0.25em
  \item Public signal \obs{$(X,D)$} determines beliefs \outcome{$\pi$}.
  \item Those beliefs determine bidder entry and expected premia.
  \item Prices then aggregate value and expected takeover surplus at each stage.
\end{itemize}
\end{alertblock}

\returnlink{main:engagement}{Slide 9: Inference, Bids, Prices}
\end{frame}

% -- B6 ----------------------------------------------------------------------
\begin{frame}{B6. Deriving the minority-gains decomposition}
\hypertarget{backup:B6}{}
\small
\begin{alertblock}{Definition}
\[
\outcome{\Delta^{\min}}(\kappa)=\E\big[\outcome{m^{R}}(\choice{a})\cdot\1\{\textup{bid}\}\big].
\]
\end{alertblock}

\begin{alertblock}{Decomposition}
Since $\outcome{m^{R}}(\choice{a})=m_0+(\tilde m-m_0)\cdot \choice{a}$ and $\E[\choice{a}\mid \obs{X},\obs{D}]=\outcome{\pi}(\obs{X},\obs{D})$,
\[
\outcome{\Delta^{\min}}(\kappa)=\underbrace{m_0\PP(\textup{bid})}_{\outcome{\Delta^{\textup{base}}}}+\underbrace{(\tilde m-m_0)\E\big[\outcome{\pi}(\obs{X},\obs{D})\cdot\1\{\textup{bid}\}\big]}_{\outcome{\Delta^{\textup{act}}}}.
\]
\end{alertblock}

The decomposition separates \emph{whether} a bid occurs (extensive margin) from the \emph{size} of the premium conditional on bidding (intensive margin).

\returnlink{main:decomp}{Slide 11: Decomposition}
\end{frame}

\begin{frame}{B7. Endpoint behavior with bounded noise (Lemma 2)}
\hypertarget{backup:B7}{}
\footnotesize

\textbf{As $\kappa\downarrow 0$ (no noise):}\;
$\PP(\obs{z}\!=\!0)\to 1$ so $\obs{X}=\choice{q}$ a.s.\ and $\outcome{P_{\textup{trade}}}(\obs{X})\to \outcome{P_{\textup{post}}}(\obs{X},\obs{D}(\obs{X}))$.

\vspace{0.3em}
Order flow perfectly reveals the action: $\obs{X}\!=\!-1$ $\Rightarrow$ Exit, $\obs{X}\!=\!0$ $\Rightarrow$ Hold/Quiet Voice, $\obs{X}\!=\!+1$ $\Rightarrow$ Public Voice. Inference is maximally sharp; execution price collapses to post-disclosure price.

\vspace{0.6em}
\textbf{As $\kappa\uparrow 1$ (maximal noise):}\;
$\obs{z}$ becomes uniform on $\{-1,0,1\}$. Because the support of $\obs{z}$ is bounded:
\begin{itemize}\itemsep0.4em
  \item Extreme flows still reveal disclosure: $\PP(\obs{D}\!=\!1\mid \obs{X}\!=\!2)=1$ and $\PP(\obs{D}\!=\!1\mid \obs{X}\in\{-2,-1\})=0$
  \item Interior flows ($\obs{X}\in\{0,1\}$) continue mixing via Bayes' rule
  \item $\outcome{\Delta^{\textup{act}}}$ does \textbf{not} vanish --- engagement survives in the Public Voice region
\end{itemize}

\begin{frame}{B19. Why use \outcome{$\Delta^{\min}$} as the welfare metric?}
\hypertarget{backup:B19}{}
\footnotesize
\textbf{Minority-shareholder object.}
With dispersed ownership, minority shareholders cannot coordinate to internalize governance benefits directly. Their main channel for capturing value from control changes is the takeover premium, consistent with the \citealp{GrossmanHart1980} free-rider logic.

\vspace{0.4em}
\textbf{Definition}
\[
\outcome{\Delta^{\min}}(\kappa)\equiv\E\big[\outcome{m^{R}}(\choice{a})\cdot\1\{\textup{bid}\}\big]
\]
This is the expected premium, weighted by bid occurrence, that minority shareholders receive ex ante.

\vspace{0.4em}
\textbf{Alternative metrics.}
Total surplus $W_{\textup{tot}}$ includes bidder surplus, so transfers cancel. It can move monotonically in $\kappa$ even when \outcome{$\Delta^{\min}$} is hump-shaped, reflecting a different policy objective.

These within-regime effects generate the inference channel. Cross-equilibrium effects also move \choice{cutoffs}, creating the general-equilibrium feedback in Propositions 5--7.

\begin{frame}{B24. Why is passive threshold crossing omitted in the baseline action set?}
\hypertarget{backup:B24}{}
\footnotesize
\textbf{Candidate omitted action:} Passive Accumulation $=(+1,0)$.

\vspace{0.4em}
\begin{alertblock}{Why it is not on-path in the baseline}
\begin{itemize}\itemsep0.3em
  \item Threshold crossing without engagement still triggers disclosure and the associated adverse-selection penalty.
  \item For high signals, the same forces that make buying attractive also make engagement attractive: expected value improvement rises while \choice{$C(s)$} falls.
  \item The appendix lemma shows that passive threshold crossing is dominated above a signal cutoff under maintained conditions; in the baseline calibration, the threshold-crossing region is therefore Public Voice.
\end{itemize}
\end{alertblock}

\begin{alertblock}{Interpretation}
The baseline action set is not ruling out passive buying by assumption. It is summarizing the on-path equilibrium object for a control-oriented activist: once she crosses the threshold in relevant high-signal states, she also chooses to engage.
\end{alertblock}

\begin{frame}{B25. Why use discrete order flow instead of continuous trading?}
\hypertarget{backup:B25}{}
\footnotesize
\begin{alertblock}{What the discrete setup buys}
\begin{itemize}\itemsep0.3em
  \item Finite order-flow states give transparent Bayes updating in both \obs{$D=0$} and \obs{$D=1$} regimes.
  \item The market-maker problem stays feed-forward once disclosure arrives after anonymous execution.
  \item The model can combine governance choice, disclosure, and takeover feedback without a price fixed point.
\end{itemize}
\end{alertblock}

\begin{alertblock}{What it gives up}
\begin{itemize}\itemsep0.3em
  \item No gradual multi-round accumulation.
  \item Bounded noise support rather than a continuous Kyle environment.
  \item No claim that the hump is a generic closed-form property of all trading protocols.
\end{itemize}
\end{alertblock}

\textbf{Interpretation:} this is a tractable first-pass theory design, not a claim that real activist trading is literally one-shot.

\begin{frame}{B26. Why does anonymous accumulation matter for Public Voice?}
\hypertarget{backup:B26}{}
\footnotesize
\begin{alertblock}{Timing wedge}
\[
\text{At } t=1:\quad \choice{\text{buy at }}\outcome{P_{\textup{trade}}}(\obs{X})
\qquad\Longrightarrow\qquad
\text{At } t=1.2:\quad \obs{D}=1 \text{ and price updates to } \outcome{P_{\textup{post}}}(\obs{X},1)
\]
\end{alertblock}

\begin{itemize}\itemsep0.35em
  \item The activist accumulates before the market fully capitalizes observed engagement.
  \item If she had to buy at the fully disclosed price, she would internalize the whole activism premium immediately.
  \item That would shrink the Public Voice region and weaken the transparency-attenuation channel in general equilibrium.
\end{itemize}

\begin{alertblock}{Bridge to the disclosure results}
Anonymous execution is what lets Public Voice survive on path in the baseline calibration. That is why the partial-equilibrium attenuation result has economic bite instead of being a purely mechanical accounting statement.
\end{alertblock}
